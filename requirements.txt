## PARTE 2 — Corrección C #17: Validación de la influencia de los índices de vegetación
## Estimado: 1.5 horas

Objetivo: cuantificar empíricamente el aporte real de los 5 índices de 
vegetación (NDVI, NDRE, CIgreen, PRI, PSRI) al desempeño final de los 
modelos, frente a la información puramente espectral de las 58 bandas.

Esto se hace mediante **ablación durante inferencia**, sin reentrenamiento.

### Paso 2.1 — Definir el protocolo de ablación

Para cada modelo final, evaluar el desempeño bajo cuatro condiciones:

(C0) Todas las features (baseline reportado en el documento): 58 bandas + 
     5 IV.
(C1) Sin índices de vegetación: las 5 posiciones correspondientes a los IV 
     se sustituyen por su VALOR PROMEDIO calculado sobre el conjunto de 
     ENTRENAMIENTO. Esto es estadísticamente equivalente a "neutralizar" 
     la contribución informativa de cada IV manteniendo la dimensionalidad 
     del vector de entrada esperado por el modelo.
(C2) Solo índices de vegetación: las 58 bandas espectrales se sustituyen 
     por sus promedios del conjunto de entrenamiento. Este es el inverso 
     de C1 y permite ver cuánta capacidad discriminativa reside 
     exclusivamente en los IV.
(C3) Sin un IV específico (5 sub-condiciones, una por cada IV): se 
     neutraliza solo el IV correspondiente. Esto permite identificar la 
     contribución individual de cada índice.

CRÍTICO: el modelo NO se reentrena. La ablación opera sobre el vector de 
entrada del conjunto de prueba durante inferencia. Los índices se 
neutralizan por el promedio de entrenamiento, NO por ceros, para evitar 
introducir valores fuera del dominio aprendido por el modelo.

Para CNN-2D: la ablación se aplica en el canal correspondiente del parche 
5×5, en todas las posiciones del parche (es decir, todo el canal del IV 
correspondiente se constantea al promedio).

class Spectral1DCNN(nn.Module):
    def __init__(self, n_classes=2, kernel_size=5, dropout=0.3):
        super().__init__()
        pad = kernel_size // 2
        self.conv1 = nn.Conv1d(1, 16, kernel_size=kernel_size, padding=pad)
        self.bn1   = nn.BatchNorm1d(16)
        self.conv2 = nn.Conv1d(16, 32, kernel_size=kernel_size, padding=pad)
        self.bn2   = nn.BatchNorm1d(32)
        self.pool  = nn.AdaptiveAvgPool1d(8)
        self.fc1   = nn.Linear(32 * 8, 64)
        self.drop  = nn.Dropout(dropout)
        self.fc2   = nn.Linear(64, n_classes)

    def forward(self, x):
        x = F.relu(self.bn1(self.conv1(x)))
        x = F.relu(self.bn2(self.conv2(x)))
        x = self.pool(x)
        x = torch.flatten(x, 1)
        x = F.relu(self.fc1(x))
        x = self.drop(x)
        return self.fc2(x)


class SpectralSpatialCNN2D(nn.Module):
    def __init__(self, n_channels=63, n_classes=2, dropout=0.3, kernel_size=3):
        super().__init__()
        pad = kernel_size // 2
        self.conv1 = nn.Conv2d(n_channels, 32, kernel_size=kernel_size, padding=pad)
        self.bn1   = nn.BatchNorm2d(32)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=kernel_size, padding=pad)
        self.bn2   = nn.BatchNorm2d(64)
        self.conv3 = nn.Conv2d(64, 128, kernel_size=kernel_size, padding=pad)
        self.bn3   = nn.BatchNorm2d(128)
        self.pool  = nn.AdaptiveAvgPool2d((2, 2))
        self.fc1   = nn.Linear(128 * 2 * 2, 128)
        self.drop  = nn.Dropout(dropout)
        self.fc2   = nn.Linear(128, n_classes)

    def forward(self, x):
        x = F.relu(self.bn1(self.conv1(x)))
        x = F.relu(self.bn2(self.conv2(x)))
        x = F.relu(self.bn3(self.conv3(x)))
        x = self.pool(x)
        x = torch.flatten(x, 1)
        x = F.relu(self.fc1(x))
        x = self.drop(x)
        return self.fc2(x)

def build_patch_dataset(cube: np.ndarray, labels_2d: np.ndarray,
                         split_2d: np.ndarray, split_value: int, patch: int = 5):
    """Extract patch dataset for CNN-2D. Excludes border pixels and NaN patches."""
    Hc, Wc, Bc = cube.shape
    r = patch // 2
    X_list, y_list = [], []
    for i in range(r, Hc - r):
        for j in range(r, Wc - r):
            if split_2d[i, j] != split_value:
                continue
            if np.isnan(labels_2d[i, j]):
                continue
            patch_cube = cube[i-r:i+r+1, j-r:j+r+1, :]
            if np.isnan(patch_cube).any():
                continue
            X_list.append(np.transpose(patch_cube, (2, 0, 1)))  # (B, p, p)
            y_list.append(labels_2d[i, j])
    return np.stack(X_list), np.array(y_list, dtype=np.int64)

def find_best_thr(y_true: np.ndarray, y_prob: np.ndarray) -> float:
    grid = np.linspace(0.10, 0.90, 161)
    return max(grid, key=lambda t: f1_score(y_true, (y_prob >= t).astype(int), average="macro"))

# CNN-1D val threshold verification
def predict_proba_1d(model, X: np.ndarray, batch_size=512) -> np.ndarray:
    model.eval()
    probs = []
    Xt = torch.from_numpy(X.astype(np.float32)).unsqueeze(1)  # (N,1,L)
    with torch.no_grad():
        for i in range(0, len(Xt), batch_size):
            logits = model(Xt[i:i+batch_size].to(DEVICE))
            prob   = torch.softmax(logits, dim=1)[:, 1].cpu().numpy()
            probs.extend(prob)
    return np.array(probs)

def predict_proba_2d(model, X: np.ndarray, batch_size=256) -> np.ndarray:
    model.eval()

        for i in range(0, len(Xt), batch_size):

            prob   = torch.softmax(logits, dim=1)[:, 1].cpu().numpy()
            probs.extend(prob)
    return np.array(probs)

METRICS = ["PR-AUC", "ROC-AUC", "F1-macro", "Recall_1", "Precision_1", "Accuracy"]

def compute_metrics(y_true, y_prob, thr):
    y_pred = (y_prob >= thr).astype(int)
    return {
        "PR-AUC":      round(average_precision_score(y_true, y_prob), 4),
        "ROC-AUC":     round(roc_auc_score(y_true, y_prob), 4),
        "F1-macro":    round(f1_score(y_true, y_pred, average="macro"), 4),
        "Recall_1":    round(recall_score(y_true, y_pred), 4),
        "Precision_1": round(precision_score(y_true, y_pred, zero_division=0), 4),
        "Accuracy":    round(accuracy_score(y_true, y_pred), 4),
    }


def ablate_and_predict_ml(pipe, X_test_orig, X_train_orig, ablate_indices):
    """
    Neutralize selected feature indices by setting them to their training mean
    in ORIGINAL (unscaled) space, then pass through the pipeline.
    """
    X_abl = X_test_orig.copy()
    for idx in ablate_indices:
        X_abl[:, idx] = X_train_orig[:, idx].mean()
    return pipe.predict_proba(X_abl)[:, 1]


def ablate_and_predict_1d(model, X_test_scaled, X_train_scaled, ablate_indices, thr, batch=512):
    """
    Neutralize in SCALED space (post-scaling mean replacement for DL models).
    For CNN-1D, features are 1D vectors.
    """
    X_abl = X_test_scaled.copy()
    for idx in ablate_indices:
        X_abl[:, idx] = X_train_scaled[:, idx].mean()
    return predict_proba_1d(model, X_abl, batch_size=batch)


def ablate_and_predict_2d(model, X_test_patches, cube_scaled, labels_bin, split_map,
                           X_train_scaled, ablate_indices, thr, patch=5, batch=256):
    """
    Neutralize a channel in the scaled cube, rebuild patches, predict.
    """
    cube_abl = cube_scaled.copy()
    H, W, B = cube_abl.shape
    for idx in ablate_indices:
        channel_train_mean = X_train_scaled[:, idx].mean()
        cube_abl[:, :, idx] = channel_train_mean
    X_abl, y_abl = build_patch_dataset(cube_abl, labels_bin, split_map, 3, patch)
    return predict_proba_2d(model, X_abl, batch_size=batch), y_abl
